### A-0 데이터 수집 (선택 — csv 이미 제공)

> `avengers_network_data.csv`가 폴더 안에 있으므로 이 단계는 선택.
> 다른 영화/드라마로 직접 만들고 싶을 때 아래 프롬프트 사용.

```
아래 CSV 형식으로 [영화명] 캐릭터 관계 네트워크 데이터를 만들어줘.
파일명: avengers_network_data.csv (UTF-8)

─── 컬럼 구조 (헤더 행 그대로) ───
Source,Target,Strength,Type,Source_Group,Source_Importance,Source_Time,Source_Image,Target_Group,Target_Importance,Target_Time,Target_Image

각 컬럼 설명:
- Source / Target        : 캐릭터 ID (영어 소문자 + 언더스코어, 예: iron_man)
- Strength               : 관계 강도 1~10 (함께 등장 장면 수 기반, 10=가장 강함)
- Type                   : ally / enemy / romantic / family 중 하나
- Source_Group / Target_Group : 캐릭터 소속 그룹 (예: Avengers, Villain, Supporting)
- Source_Importance / Target_Importance : 화면 비중 1~10 (주연=8~10, 조연=4~7, 단역=1~3)
- Source_Time / Target_Time  : 화면 등장 시간 "분:초" 형식 (예: 14:30)
- Source_Image / Target_Image : Wikipedia 또는 Wikia 직접 접근 가능한 이미지 URL

데이터 기준:
- 같은 장면에 2회 이상 등장한 쌍만 포함
- 총 60~90행 목표
- 이미지 URL은 실제 접근 가능한 URL로 (Wikia static URL 권장)
```

✅ 확인: CSV 저장 → 헤더 12개 컬럼 → 60행 이상

```
function calculateLayout(selId) {
  const padding = Math.min(W,H) * 0.16;
  const R = (H/2) - padding;
  const cy = H/2;
  let cx = W*0.2 + R;
  cx = Math.max(cx, padding+R); cx = Math.min(cx, W-padding-R);

  const visibleOrder = circleOrder.filter(n => filteredIds.has(n.id));
  const N = visibleOrder.length;
  const selIdx = visibleOrder.findIndex(n => n.id === selId);
  const slotAngle = N > 0 ? (2*Math.PI)/N : 0;

  visibleOrder.forEach((n,i) => {
    const relIdx = ((i - selIdx) + N) % N;
    const angle = Math.PI + relIdx * slotAngle;   // 선택 노드 → 9시 방향(π)
    pos[n.id] = { x: cx + Math.cos(angle)*R, y: cy + Math.sin(angle)*R, ... };
  });
  // 필터에서 제외된 노드: { x:cx, y:cy, hidden:true } (중앙으로 숨김)
}

─── 곡선 엣지 (quadratic bezier) — 이 코드 그대로 ───
function makeCurve(sx,sy,tx,ty) {
  const dx=tx-sx, dy=ty-sy, dist=Math.sqrt(dx*dx+dy*dy);
  if (dist<1) return "M"+sx+","+sy+"L"+tx+","+ty;
  const nx=-dy/dist, ny=dx/dist;
  const curvature = dist * 0.18;        // ← 이 계수 그대로
  const mx=(sx+tx)/2 + nx*curvature;
  const my=(sy+ty)/2 + ny*curvature;
  return "M"+sx+","+sy+" Q"+mx+","+my+" "+tx+","+ty;
}
```

```
function latLngToVec3(lat, lng, r, h) {
  var phi = (90 - lat) * toRAD;
  var theta = (lng + 180) * toRAD;
  var rad = r + (h || 0);
  return new THREE.Vector3(
    -rad * Math.sin(phi) * Math.cos(theta),
    rad * Math.cos(phi),
    rad * Math.sin(phi) * Math.sin(theta)
  );
}
```

```
function createGlowTexture(color, softness) {
  var canvas = document.createElement('canvas');
  canvas.width = 64; canvas.height = 64;
  var ctx = canvas.getContext('2d');
  var gradient = ctx.createRadialGradient(32, 32, 0, 32, 32, 32);

  gradient.addColorStop(softness || 0.3, color || 'rgba(255,255,255,0.5)');
  gradient.addColorStop(1, 'rgba(255,255,255,0)');
  ctx.fillStyle = gradient;
  ctx.fillRect(0, 0, 64, 64);
  return new THREE.CanvasTexture(canvas);
}
```

```
  return n.toString();
}
```
