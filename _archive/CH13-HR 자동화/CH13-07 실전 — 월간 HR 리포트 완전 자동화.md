## 직접 해보기

> **💡 이 클립만 따로 시작한다면** — 먼저 실행해서 데이터 생성
> ```bash
> mkdir -p ~/hr-project/data/raw ~/hr-project/data/output ~/hr-project/charts ~/hr-project/reports && cd ~/hr-project
> python3 -c "
> import pandas as pd, random; from datetime import date, timedelta; random.seed(42)
> depts=['개발','마케팅','영업','운영','인사']; levels=['주니어','시니어','리드']
> emps,perf,attr=[],[],[]
> for eid in range(500):
>     dept=random.choice(depts); level=random.choices(levels,weights=[5,3,2])[0]
>     hire=date(2020,1,1)+timedelta(days=random.randint(0,1460))
>     emps.append({'employee_id':f'E{eid:03d}','department':dept,'level':level,'hire_date':str(hire)})
>     for q in ['2023-Q1','2023-Q2','2023-Q3','2023-Q4']:
>         s=round(max(1.0,min(5.0,random.gauss(3.2,0.8))),1)
>         perf.append({'employee_id':f'E{eid:03d}','period':q,'score':s,'rating':'우수' if s>=4 else '보통' if s>=2.5 else '미흡'})
>     if random.random()<0.15:
>         attr.append({'employee_id':f'E{eid:03d}','leave_date':str(date(2023,1,1)+timedelta(days=random.randint(0,365))),'reason':random.choice(['이직','개인사유','계약만료'])})
> pd.DataFrame(emps).to_csv('data/raw/employees.csv',index=False)
> pd.DataFrame(perf).to_csv('data/raw/performance.csv',index=False)
> pd.DataFrame(attr).to_csv('data/raw/attrition.csv',index=False)
> print(f'생성 완료: 직원 {len(emps)}명')
> "
> ```



**준비**
